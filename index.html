function parseNumber(val) { if (val === null || val === undefined) return 0; // Преобразуем любое значение (число, объект, строку) в безопасный формат const str = String(val).replace(/\s+/g, '').replace(',', '.'); const num = parseFloat(str); return isNaN(num) ? 0 : num; } function handleFileSelect(e) { const file = e.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = (event) => { try { const data = new Uint8Array(event.target.result); const workbook = XLSX.read(data, { type: 'array' }); const firstSheet = workbook.Sheets[workbook.SheetNames[0]]; // Получаем массив данных с сырыми значениями const jsonData = XLSX.utils.sheet_to_json(firstSheet, { header: 1, defval: '' }); if (!jsonData || jsonData.length === 0) return alert("Файл пуст или не содержит данных!"); // Автоматический поиск строки-заголовка (где найдем "наименование", "код", "товар" или больше 2 заправленных ячеек) let headerRowIdx = jsonData.findIndex(row => { if (!Array.isArray(row)) return false; const rowStr = row.join(' ').toLowerCase(); return rowStr.includes('наимен') || rowStr.includes('товар') || rowStr.includes('номенкл') || row.filter(Boolean).length >= 3; }); if (headerRowIdx === -1) headerRowIdx = 0; rawHeaders = jsonData[headerRowIdx].map(h => String(h || '').trim()); rawExcelData = jsonData.slice(headerRowIdx + 1); populateDropdowns(rawHeaders); document.getElementById('mappingPanel').classList.remove('hidden'); speak("Файл прочитан. Подтвердите сопоставление колонок."); } catch (err) { console.error(err); alert("Ошибка при чтении файла Excel. Убедитесь, что файл не поврежден."); } }; reader.readAsArrayBuffer(file); } function applyMapping() { const nameIdx = parseInt(document.getElementById('mapName').value); const codeIdx = parseInt(document.getElementById('mapCode').value); const unitIdx = parseInt(document.getElementById('mapUnit').value); const priceIdx = parseInt(document.getElementById('mapPrice').value); const accQtyIdx = parseInt(document.getElementById('mapAccQty').value); if (nameIdx === -1) return alert("Укажите колонку с Наименованием!"); const tbody = document.getElementById('tableBody'); tbody.innerHTML = ''; let count = 0; rawExcelData.forEach(row => { if (!Array.isArray(row)) return; const nameVal = row[nameIdx] !== undefined ? String(row[nameIdx]).trim() : ''; if (!nameVal) return; count++; const codeVal = codeIdx !== -1 && row[codeIdx] !== undefined ? String(row[codeIdx]).trim() : ''; const unitVal = unitIdx !== -1 && row[unitIdx] !== undefined ? String(row[unitIdx]).trim() : 'шт'; const priceVal = priceIdx !== -1 ? parseNumber(row[priceIdx]) : 0; const accQtyVal = accQtyIdx !== -1 ? parseNumber(row[accQtyIdx]) : 0; const factQtyVal = 0; const accSumVal = priceVal * accQtyVal; const factSumVal = priceVal * factQtyVal; const diff = factQtyVal - accQtyVal; const defQty = diff < 0 ? Math.abs(diff) : 0; const defSum = defQty * priceVal; const surQty = diff > 0 ? diff : 0; const surSum = surQty * priceVal; const tr = document.createElement('tr'); tr.innerHTML = ` ${count} ${factSumVal.toFixed(2)} ${accSumVal.toFixed(2)} ${defQty} ${defSum.toFixed(2)} ${surQty} ${surSum.toFixed(2)} `; tbody.appendChild(tr); }); recalcTotals(); document.getElementById('mappingPanel').classList.add('hidden'); speak(`Загружено ${count} позиций`); }
